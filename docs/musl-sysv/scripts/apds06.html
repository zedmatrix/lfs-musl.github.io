D.6. /etc/rc.d/init.d/gen-ifconfig 

#!/bin/bash
### BEGIN INIT INFO
# Provides:            gen-ifconfig
# Required-Start:      $local_fs $network
# Required-Stop:       $network
# Default-Start:       S
# Default-Stop:
# Short-Description:   generates dynamic ifconfig
# Description:         generates dynamic ifconfig on load from iface
### END INIT INFO

. /lib/lsb/init-functions

CONFIG_DIR="/etc/sysconfig"

write_config() {
    local iface="$1" service="$2" extra="$3"
    local file="$CONFIG_DIR/ifconfig.$iface"
    log_success_msg "Creating $file"
    cat > "$file" <<EOF
ONBOOT="yes"
IFACE="$iface"
SERVICE="$service"
DHCP_START="-b -q -h $HOSTNAME"
DHCP_STOP="-k"
$extra
EOF
}

log_info_msg "Generating dynamic interface config..."

for path in /sys/class/net/*; do
    iface="${path##*/}"
    case "$iface" in
        e*) write_config "$iface" "dhcpcd" ;;
        w*) write_config "$iface" "wpa" 'WPA_ARGS="" WPA_SERVICE="dhcpcd"' ;;
        *) continue ;;
    esac
done

exit 0
# End of gen-ifconfig
